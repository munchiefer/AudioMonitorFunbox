// 1.4.2 with FIXED per-setting gains (no auto-adjust): loudness of each effect vs the dry voice (ITU-R BS.1770),
// at a normal (-26 LUFS), quiet (-35) and loud (-18) voice; the gain never moves while talking; Volume; walkie
// sounds; regressions. Synthetic voice + light room noise, offline: no devices, no settings.
using NAudio.Wave;
using AudioMonitor;

const int Rate = 48000;
int fails = 0;
void Check(bool ok, string what) { Console.WriteLine($"{(ok ? "PASS" : "FAIL")}  {what}"); if (!ok) fails++; }

float[] Voice(double seconds, int seed, List<(int From, int To)>? pauses = null)
{
    var rng = new Random(seed);
    int n = (int)(Rate * seconds);
    var y = new float[n];
    var vowels = new (double F1, double F2, double F3)[] { (730, 1090, 2440), (530, 1840, 2480), (270, 2290, 3010), (570, 840, 2410), (300, 870, 2240) };
    int pos = 0; double phase = 0, tilt = 0; var res = new double[3, 2];
    while (pos < n)
    {
        int syl = (int)(Rate * (0.18 + rng.NextDouble() * 0.14));
        int gap = (int)(Rate * (rng.NextDouble() < 0.2 ? 0.3 + rng.NextDouble() * 0.2 : 0.04 + rng.NextDouble() * 0.1));
        var v = vowels[rng.Next(vowels.Length)];
        double f0 = 105 + rng.NextDouble() * 45;
        int fric = rng.NextDouble() < 0.35 ? Rate * 6 / 100 : 0;
        double hpPrev = 0, hpOut = 0;
        for (int i = 0; i < syl + fric && pos < n; i++, pos++)
        {
            double sample;
            if (i < fric) { double noise = rng.NextDouble() * 2 - 1; hpOut = 0.7 * (hpOut + noise - hpPrev); hpPrev = noise; sample = hpOut * 0.25 * Math.Sin(Math.PI * i / fric); }
            else
            {
                int k = i - fric;
                double env = Math.Min(1, Math.Min(k / (0.02 * Rate), (syl - k) / (0.03 * Rate)));
                phase += f0 * (1 + 0.08 * Math.Sin(2 * Math.PI * k / syl)) / Rate; if (phase >= 1) phase -= 1;
                tilt += (1 - 2 * phase - tilt) * 0.15;
                double sum = 0; double[] fs = { v.F1, v.F2, v.F3 }, gains = { 1.0, 0.5, 0.25 };
                for (int r = 0; r < 3; r++)
                {
                    double bw = 80 + fs[r] * 0.05, rr = Math.Exp(-Math.PI * bw / Rate), th = 2 * Math.PI * fs[r] / Rate;
                    double o = (1 - rr) * tilt + 2 * rr * Math.Cos(th) * res[r, 0] - rr * rr * res[r, 1];
                    res[r, 1] = res[r, 0]; res[r, 0] = o; sum += o * gains[r];
                }
                sample = sum * env;
            }
            y[pos] = (float)sample;
        }
        if (gap > Rate / 4 && pos + gap < n) pauses?.Add((pos + Rate / 10, pos + gap - Rate / 20));
        pos += gap;
    }
    return y;
}
double Lufs(float[] x, int from = 0, int to = -1)
{
    if (to < 0) to = x.Length;
    var shelf = new PedalChain.Biquad().HighShelf(Rate, 1681.97, 4.0); var hp = new PedalChain.Biquad().HighPass(Rate, 38.13, 0.5);
    var k = new double[to - from];
    for (int i = from; i < to; i++) k[i - from] = hp.Process(shelf.Process(x[i]));
    int block = Rate * 4 / 10, hop = Rate / 10; var z = new List<double>();
    for (int s = 0; s + block <= k.Length; s += hop) { double m = 0; for (int i = s; i < s + block; i++) m += k[i] * k[i]; z.Add(m / block); }
    double L(double ms) => -0.691 + 10 * Math.Log10(ms + 1e-20);
    var abs = z.Where(m => L(m) > -70).ToList(); if (abs.Count == 0) return -99;
    double rel = L(abs.Average()) - 10;
    return L(abs.Where(m => L(m) > rel).Average());
}
double RmsDb(float[] x, IEnumerable<(int From, int To)> spans) { double s = 0; long c = 0; foreach (var (a, b) in spans) for (int i = a; i < b; i++) { s += x[i] * x[i]; c++; } return 10 * Math.Log10(s / Math.Max(1, c) + 1e-20); }

var pauses = new List<(int, int)>();
var speech = Voice(16, 7, pauses);   // a different voice than the built-in calibration voice (different seed / length)
var rngNoise = new Random(3);
float[] Noisy(float[] voice, double voiceLufs)
{
    double g = Math.Pow(10, (voiceLufs - Lufs(voice)) / 20);
    return voice.Select(v => (float)(v * g + (rngNoise.NextDouble() * 2 - 1) * 0.0006)).ToArray();
}
int start = Rate * 2;   // effect switched on after 2 s
(float[] Out, List<float> Gains) Run(VoiceFxSettings fx, float[] mono, Action<VoiceEffects, int>? perBlock = null)
{
    var effects = new VoiceEffects(new ArraySource(mono), new VoiceFxSettings { Enabled = false, Volume = fx.Volume });
    var output = new float[mono.Length]; var gains = new List<float>(); var block = new float[960];
    for (int i = 0, b = 0; i < mono.Length; i += 480, b++)
    {
        if (i == start) { effects.Update(fx); effects.WaitForCalibration(); }
        perBlock?.Invoke(effects, b);
        int n = Math.Min(480, mono.Length - i);
        effects.Read(block, 0, n * 2);
        for (int j = 0; j < n; j++) output[i + j] = block[j * 2];
        if (i > start + Rate / 2) gains.Add(effects.Loudness.GainDb);
    }
    return (output, gains);
}

var configs = new List<(string Name, Action<VoiceFxSettings> Set, bool Linear)>
{
    ("Pitch +7 (Chipmunk)", f => { f.PitchOn = true; f.PitchSemitones = 7; }, true),
    ("Pitch -5 (Deep voice)", f => { f.PitchOn = true; f.PitchSemitones = -5; }, true),
    ("Underwater 50", f => { f.UnderwaterOn = true; f.UnderwaterDepth = 50; }, true),
    ("Underwater 100", f => { f.UnderwaterOn = true; f.UnderwaterDepth = 100; }, true),
    ("Echo 300 ms", f => { f.EchoOn = true; f.EchoMs = 300; }, true),
    ("Reverb 100", f => { f.ReverbOn = true; f.ReverbSize = 100; }, true),
    ("Cave preset", f => { f.ReverbOn = true; f.ReverbSize = 90; f.EchoOn = true; f.EchoMs = 400; }, true),
    ("Villain preset", f => { f.PitchOn = true; f.PitchSemitones = -4; f.ReverbOn = true; f.ReverbSize = 60; }, true),
    ("Ghost 100", f => { f.GhostOn = true; f.GhostAmount = 100; }, true),
    ("Demon 100", f => { f.DemonOn = true; f.DemonDepth = 100; }, false),
    ("Telephone 40", f => { f.TelephoneOn = true; f.TelephoneGrit = 40; }, false),
    ("Radio 0", f => { f.RadioOn = true; f.RadioDrive = 0; }, false),
    ("Radio 100", f => { f.RadioOn = true; f.RadioDrive = 100; }, false),
    ("Mission control", f => { f.RadioOn = true; f.RadioDrive = 55; f.WalkieTalkie = true; }, false),
};
Console.WriteLine($"{"effect (vs dry mic)",-24} {"normal voice",13} {"quiet voice",13} {"loud voice",13}");
double worstNormal = 0, worstLinear = 0, worstSpread = 0;
foreach (var (name, set, linear) in configs)
{
    var cells = new List<string>();
    foreach (double level in new[] { -26.0, -35.0, -18.0 })
    {
        var dry = Noisy(speech, level);
        var fx = new VoiceFxSettings { Enabled = true }; set(fx);
        var (wet, gains) = Run(fx, dry);
        double d = Lufs(wet, start) - Lufs(dry, start);
        cells.Add($"{d,+8:+0.0;-0.0} dB");
        worstSpread = Math.Max(worstSpread, gains.Max() - gains.Min());
        if (level == -26 && !name.StartsWith("Mission")) worstNormal = Math.Max(worstNormal, Math.Abs(d));
        if (linear) worstLinear = Math.Max(worstLinear, Math.Abs(d));
    }
    Console.WriteLine($"{name,-24} {cells[0],13} {cells[1],13} {cells[2],13}");
}
Check(worstSpread == 0, "no auto-adjust: each effect's gain stays exactly fixed while talking");
Check(worstNormal <= 1.5, $"normal speaking level: every effect within {worstNormal:0.0} dB of the dry mic");
Check(worstLinear <= 1.5, $"filters/pitch/echo/reverb/ghost: within {worstLinear:0.0} dB at any speaking level");

// Same setting -> same gain, every time (and across instances).
{
    var fx = new VoiceFxSettings { Enabled = true, RadioOn = true, RadioDrive = 70 };
    var a = new VoiceEffects(new ArraySource(new float[480]), fx).Loudness.CalibDb;
    var b = new VoiceEffects(new ArraySource(new float[480]), fx).Loudness.CalibDb;
    Check(a == b, $"a setting always gets the same fixed gain ({a:+0.0;-0.0} dB for Radio 70)");
}

// Volume: 100 % = untouched (effects off), 50 % = -6 dB, 200 % = +6 dB.
{
    var dry = Noisy(speech, -26);
    var (full, _) = Run(new VoiceFxSettings { Enabled = false, Volume = 100 }, dry);
    double maxDiff = 0; for (int i = 0; i < dry.Length; i++) maxDiff = Math.Max(maxDiff, Math.Abs(full[i] - dry[i]));
    Check(maxDiff == 0, "Volume 100 % with effects off: the mic passes through bit-for-bit unchanged");
    double half = Lufs(Run(new VoiceFxSettings { Enabled = false, Volume = 50 }, dry).Out, start) - Lufs(dry, start);
    double twice = Lufs(Run(new VoiceFxSettings { Enabled = false, Volume = 200 }, dry).Out, start) - Lufs(dry, start);
    Check(Math.Abs(half + 6.02) < 0.3 && Math.Abs(twice - 6.02) < 0.5, $"Volume 50 % = {half:+0.0;-0.0} dB, 200 % = {twice:+0.0;-0.0} dB");
}

// Walkie-talkie: push-to-talk keys the radio even with "Voice effects on" ticked; fixed static level.
{
    var dry = Noisy(speech, -26);
    var latched = new VoiceFxSettings { Enabled = true, RadioOn = true, RadioDrive = 55, WalkieTalkie = true };
    int press = (start + Rate * 3) / 480, release = (start + Rate * 7) / 480;
    var (wet, _) = Run(latched, dry, (e, b) => { if (b == press) e.SetPushToTalk(true); if (b == release) e.SetPushToTalk(false); });
    double voiceRms = Math.Sqrt(Math.Pow(10, RmsDb(dry, new[] { (start, start + 2 * Rate) }) / 10));
    double burst = Math.Sqrt(Math.Pow(10, RmsDb(wet, new[] { (press * 480, press * 480 + Rate * 6 / 100) }) / 10));
    double tail = Math.Sqrt(Math.Pow(10, RmsDb(wet, new[] { (release * 480 + Rate / 100, release * 480 + Rate * 25 / 100) }) / 10));
    Check(burst > voiceRms * 0.3, $"push-to-talk press: \"chk-shh\" ({20 * Math.Log10(burst / voiceRms):+0;-0} dB vs a normal voice)");
    Check(tail > voiceRms * 0.3, $"release: \"kshhh\" static ({20 * Math.Log10(tail / voiceRms):+0;-0} dB vs a normal voice)");
    var quiet = Noisy(speech, -38);
    var (wq, _) = Run(latched, quiet);
    int longPauses = 0, tails = 0;
    foreach (var (a, b) in pauses.Where(p => p.Item1 > start + Rate))
    {
        longPauses++;
        if (Math.Sqrt(Math.Pow(10, RmsDb(wq, new[] { (a, Math.Min(b, a + Rate / 5)) }) / 10)) > 1e-3) tails++;
    }
    Check(tails >= longPauses * 0.8, $"quiet voice, no push-to-talk: static after {tails} of {longPauses} sentence ends");
}

// Regressions: push-to-talk turns effects on only while held; switching doesn't click.
{
    var input = Enumerable.Range(0, Rate * 3).Select(i => 0.2f * (float)Math.Sin(2 * Math.PI * 200 * i / Rate)).ToArray();
    var e = new VoiceEffects(new ArraySource(input), new VoiceFxSettings { Enabled = false, PitchOn = true, PitchSemitones = 12 });
    var y = new float[input.Length]; var block = new float[960];
    for (int i = 0, b = 0; i < input.Length; i += 480, b++)
    {
        if (b == 50) e.SetPushToTalk(true); if (b == 150) e.SetPushToTalk(false);
        e.Read(block, 0, 960); for (int j = 0; j < 480; j++) y[i + j] = block[j * 2];
    }
    double G(int from, double hz) { double w = 2 * Math.PI * hz / Rate, c = 2 * Math.Cos(w), s1 = 0, s2 = 0; for (int i = 0; i < 16384; i++) { double win = 0.5 - 0.5 * Math.Cos(2 * Math.PI * i / 16383); double s0 = y[from + i] * win + c * s1 - s2; s2 = s1; s1 = s0; } return Math.Sqrt(Math.Max(0, s1 * s1 + s2 * s2 - c * s1 * s2)) * 2 / 8192; }
    Check(G(4800, 200) > 0.18 && G(30000, 200) < 0.03 && G(30000, 400) > 0.05 && G(100000, 200) > 0.18, "push-to-talk: normal voice, effects only while held, normal voice again");
    double worst = 0, steady = 0;
    for (int i = 24001; i < 24000 + 1440; i++) worst = Math.Max(worst, Math.Abs(y[i] - y[i - 1]));
    for (int i = 40001; i < 46000; i++) steady = Math.Max(steady, Math.Abs(y[i] - y[i - 1]));
    Check(worst <= steady * 1.2, $"switching on mid-word: no click ({worst:0.000} vs steady {steady:0.000})");
}

Console.WriteLine(fails == 0 ? "ALL PASS" : $"{fails} FAILED");

sealed class ArraySource(float[] mono) : ISampleProvider
{
    int pos;
    public WaveFormat WaveFormat { get; } = WaveFormat.CreateIeeeFloatWaveFormat(48000, 2);
    public int Read(float[] b, int off, int count)
    {
        for (int i = 0; i < count / 2; i++) { float v = pos < mono.Length ? mono[pos] : 0; pos++; b[off + 2 * i] = b[off + 2 * i + 1] = v; }
        return count;
    }
}
